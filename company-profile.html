/* 1. رفع شيت Excel واستخراج أسماء الأعمدة ديناميكياً */ function uploadEmployeeExcel() { const fileInput = document.getElementById('excelFileInput'); if (!fileInput.files.length) return alert('برجاء اختيار ملف Excel أولاً'); const reader = new FileReader(); reader.onload = function(e) { const data = new Uint8Array(e.target.result); const workbook = XLSX.read(data, { type: 'array' }); const firstSheet = workbook.Sheets[workbook.SheetNames[0]]; const jsonData = XLSX.utils.sheet_to_json(firstSheet); if (jsonData.length > 0) { globalEmployees = jsonData; // استخراج عناوين جميع الأعمدة الموجودة بالشيت تلقائياً columnKeys = Object.keys(jsonData[0]); visibleColumns = {}; columnKeys.forEach(k => visibleColumns[k] = true); db.ref(`companyProfiles/${activeUser}/employees`).set(jsonData); db.ref(`companyProfiles/${activeUser}/columnKeys`).set(columnKeys); renderAdvancedTable(); alert('تم رفع الشيت واستخراج الأعمدة بنجاح!'); } }; reader.readAsArrayBuffer(fileInput.files[0]); } /* 2. تعديل اسم العمود (Rename Column) */ function editColumnName(oldKey) { const newKey = prompt('أدخل الاسم الجديد للعمود:', oldKey); if (!newKey || newKey.trim() === '' || newKey === oldKey) return; const trimmedNewKey = newKey.trim(); if (columnKeys.includes(trimmedNewKey)) { return alert('هذا الاسم موجود بالفعل!'); } // تحديث مصفوفة الأعمدة const index = columnKeys.indexOf(oldKey); if (index !== -1) { columnKeys[index] = trimmedNewKey; } // نقل حالة الظهور والفلتر visibleColumns[trimmedNewKey] = visibleColumns[oldKey]; delete visibleColumns[oldKey]; if (columnFilters[oldKey]) { columnFilters[trimmedNewKey] = columnFilters[oldKey]; delete columnFilters[oldKey]; } // تعديل مفتاح العمود داخل جميع بيانات الموظفين globalEmployees.forEach(emp => { if (emp.hasOwnProperty(oldKey)) { emp[trimmedNewKey] = emp[oldKey]; delete emp[oldKey]; } }); renderAdvancedTable(); saveEmployeeData(); } /* 3. إعادة بناء الجدول مع إصلاح مشكلة إخفاء العنوان والداتا */ function renderAdvancedTable() { // بناء شريط الـ Checkboxes للأعمدة const toggleContainer = document.getElementById('columnToggleList'); toggleContainer.innerHTML = ''; columnKeys.forEach(key => { if (visibleColumns[key] === undefined) visibleColumns[key] = true; toggleContainer.innerHTML += ` `; }); // بناء ترويسة الجدول والفلتر (فقط للأعمدة المفعلة visibleColumns) const headRow = document.getElementById('tableHeadRow'); const filterRow = document.getElementById('tableFilterRow'); headRow.innerHTML = ''; filterRow.innerHTML = ''; columnKeys.forEach((key, idx) => { if (visibleColumns[key]) { headRow.innerHTML += ` ${key}
`; filterRow.innerHTML += ` `; } }); renderTableBody(); } /* 4. عرض محتوى الخلايا للـ Active Columns فقط */ function renderTableBody() { const tbody = document.getElementById('empTableBody'); tbody.innerHTML = ''; const activeKeys = columnKeys.filter(k => visibleColumns[k]); const filteredEmps = globalEmployees.filter(emp => { return Object.keys(columnFilters).every(key => { if (!columnFilters[key] || !visibleColumns[key]) return true; const val = String(emp[key] || '').toLowerCase(); return val.includes(columnFilters[key]); }); }); if (filteredEmps.length === 0) { tbody.innerHTML = `لا توجد بيانات تطابق الفلترة.`; return; } filteredEmps.forEach((emp, rIdx) => { let rowHtml = ''; columnKeys.forEach(key => { if (visibleColumns[key]) { rowHtml += ` ${emp[key] || ''} `; } }); rowHtml += ''; tbody.innerHTML += rowHtml; }); }
